根據您提供的錯誤訊息，我發現了兩個主要問題：

### 1. SerialWorker 初始化錯誤
```
錯誤: 執行指令失敗: SerialWorker.__init__() missing 9 required positional arguments: 'com', 'cmd_list', 'end_str', 'timeout', 'on_data', 'on_status', 'on_progress', 'on_finish', and 'stop_event'
```

**原因**：SerialWorker 需要 9 個必要參數，但我們只傳遞了 1 個參數。

### 2. 輸出顯示問題

```python
def execute_command_thread(self, command, com_port, timeout, transport_mode):
    """在新執行緒中執行指令"""
    try:
        # 更新狀態
        self.status_label.config(text="執行中...")
        self.execute_button.config(state='disabled')
        self.update_connection_light('yellow')
        
        # 提取實際的 COM 口名稱
        actual_com_port = extract_com_port_name(com_port)
        
        # 創建串口連接
        ser = serial.Serial(actual_com_port, 115200, timeout=timeout)
        self.serial_connection = ser
        
        # 發送指令
        ser.write(f"{command}\r\n".encode('utf-8'))
        
        # 讀取回應
        response = ""
        start_time = time.time()
        
        while time.time() - start_time < timeout:
            if ser.in_waiting:
                data = ser.read(ser.in_waiting).decode('utf-8', errors='ignore')
                response += data
                time.sleep(0.1)
            else:
                time.sleep(0.1)
        
        # 關閉串口
        ser.close()
        self.serial_connection = None
        
        # 顯示結果
        if response:
            self.append_output(response)
        else:
            self.append_output("指令執行完成，無回應")
        
    except Exception as e:
        self.show_error(f"執行指令失敗: {e}")
    finally:
        self.command_finished()
```

```python
def append_output(self, text):
    """添加輸出文字（垂直排列）"""
    # 將文字按行分割並逐行顯示
    lines = text.split('\n')
    for line in lines:
        if line.strip():  # 只顯示非空行
            self.output_text.insert(tk.END, f"{line}\n")
    self.output_text.see(tk.END)
```

```python
def on_close(self):
    """關閉時的清理工作"""
    try:
        # 關閉所有 COM Port
        if self.serial_connection:
            self.serial_connection.close()
        
        # 清除快取
        self.command_history.clear()
        self.command_groups.clear()
        
        # 保存設定

    except Exception as e:
        print(f"清理手動輸入指令模組失敗: {e}")
```
