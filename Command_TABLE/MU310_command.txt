==MU310==
build time=cat /etc/pega_fwinfo
EXIT=exit
Login=exitt==>root==>oelinux123==>cat /etc/pega_fwinfo

booting_state=uci -P /var/state get pega.led.booting_state
get_model=atcli at+cgmm
fw_version=uci -P /var/state get pega.fw.cur_ver
check Button=/etc/diag.sh resetBtnStatusGet
run_mfg_script=manufacture.sh
set_pega_model=fw_setenv pega_model MU310
get_pega_model=fw_printenv -n pega_model

set_eth_mac=fw_setenv eth0_addr 11:22:33:44:55:66
get_eth_mac=fw_printenv -n eth0_addr
set_wifi_mac=fw_setenv wifi0_addr 22:33:44:55:66:77
get_wifi_mac=fw_printenv -n wifi0_addr
set_hwver=fw_setenv hwver SR
get_hwver=fw_printenv -n hwver
set_mb_sn=fw_setenv mb_sn MU310_1234
get_mb_sn=fw_printenv -n mb_sn

get_imei=atcli at+gsn
get_fw_ver2=atcli at+cgmr
get_eth_info=ethtool eth0

check_sim/CCID/CIMI=atcli at+cpin?==>atcli at+ccid==>atcli at+cimi

set_usb_mode=atcli 'AT+QCFG="usbcfg",0x2C7C,0x0801,2,1,1,0,0,1,0'
get_usb_mode=atcli 'AT+QCFG="usbcfg"'
set_mfg_flag=fw_setenv mfg 1
get_mfg_flag=fw_printenv -n mfg


LED signal OFF=echo 0 > /sys/class/leds/signal\:blue/brightness==>echo 0 > /sys/class/leds/signal\:green/brightness==>echo 0 > /sys/class/leds/signal\:red/brightness
LED status OFF=echo 0 > /sys/class/leds/status\:blue/brightness==>echo 0 > /sys/class/leds/status\:red/brightness==>echo 0 > /sys/class/leds/status\:green/brightness

Check signal LED GREEN =echo 255 > /sys/class/leds/signal\:green/brightness
Check signal LED RED=echo 255 > /sys/class/leds/signal\:red/brightness
Check signal LED BLUE=echo 255 > /sys/class/leds/signal\:blue/brightness

Check status LED GREEN =echo 255 > /sys/class/leds/status\:green/brightness
Check status LED RED=echo 255 > /sys/class/leds/status\:red/brightness
Check status LED BLUE=echo 255 > /sys/class/leds/status\:blue/brightness
